      today_pnl: Math.round((price - prev_close) * shares),

      cost: cost,
      mv: mv,
      pnl: pnl,

    };
  });

      mv: mv,
      pnl: pnl,
      pnl_pct: cost > 0 ? Number((pnl / cost * 100).toFixed(2)) : 0
    };
  });

  // 同步從「證券資料總表」動態讀取【已實現投資損益】(自動定位 2026/2025/2024 年份標題欄，不怕往左插入新年度位移)
  let quant = null;
  try {
    const srcSs = SpreadsheetApp.openById(CONFIG.SOURCE_SPREADSHEET_ID);
    const realSheet = srcSs.getSheetByName('已實現投資損益');
    if (realSheet) {
      const maxCol = realSheet.getLastColumn();
      const row3 = realSheet.getRange(3, 1, 1, maxCol).getValues()[0];
      const row4 = realSheet.getRange(4, 1, 1, maxCol).getValues()[0];
      const monthsGrid = realSheet.getRange(5, 1, 12, maxCol).getValues();
      const monthly_history = {};
      ['2026', '2025', '2024'].forEach(yr => {
        const colIdx = row3.findIndex(v => String(v).trim().replace(/\.0$/, '') === yr);
        if (colIdx !== -1) {
          const hasFut = String(row4[colIdx + 3] || '').trim() === '期貨';
          monthly_history[yr] = monthsGrid.map((mRow, mIdx) => {
            const divVal = Math.round(Number(mRow[colIdx + 1]) || 0);
            const capVal = Math.round((Number(mRow[colIdx + 2]) || 0) + (hasFut ? (Number(mRow[colIdx + 3]) || 0) : 0));
            return { month: mIdx + 1, div: divVal, cap: capVal, total: divVal + capVal };
          });
        }
      });
      quant = { monthly_history: monthly_history };
    }
  } catch (e) {}
  return { stocks: stocks, db_rows: db_rows, quant: quant, updatedAt: Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy/MM/dd HH:mm:ss') };
}

/**
 * 從「證券資料總表」【總表】與【資料庫】同步最新庫存股數與成本
 */
function syncHoldingsFromMaster() {
  const srcSs = SpreadsheetApp.openById(CONFIG.SOURCE_SPREADSHEET_ID);
  const dstSs = SpreadsheetApp.openById(CONFIG.STOCK_SPREADSHEET_ID);

  const srcMain = srcSs.getSheetByName('總表');
  const lastRow = srcMain.getLastRow();
  const rawMain = srcMain.getRange(8, 2, Math.max(1, lastRow - 7), 13).getValues();
  const validStocks = rawMain.filter(r => r[0] !== '' && r[0] !== null && Number(r[7]) > 0);

  const dstMain = dstSs.getSheetByName(CONFIG.MAIN_SHEET_NAME);
  const existingRows = dstMain.getRange(8, 1, 43, 12).getValues();
  const priceMap = {};
  existingRows.forEach(r => {
    const code = String(r[0]).trim().replace(/^'/, '');
    if (code) priceMap[code] = { prevClose: Number(r[10]) || 0, price: Number(r[11]) || 0 };
  });

  validStocks.forEach((r, idx) => {
    const targetRow = 8 + idx;
    let code = String(r[0]).trim().replace(/\.0$/, '');
    const name = String(r[1]).trim();
    const isOtc = CONFIG.OTC_CODES.indexOf(code) !== -1;
    const market = isOtc ? '上櫃 TPEx' : '上市 TWSE';
    const stype = String(r[12] || '').trim();
    const ch = Number(r[2]) || 0;
    const cathay = Number(r[3]) || 0;
    const yuanta = Number(r[4]) || 0;
    const fubon = Number(r[5]) || 0;
    const kgi = Number(r[6]) || 0;
    const totShares = Number(r[7]) || 0;
    const totCost = Number(r[8]) || 0;
    const totMv = Number(r[9]) || 0;
    const inferredPrice = totShares > 0 ? Number((totMv / totShares).toFixed(2)) : 0;

    dstMain.getRange(targetRow, 1, 1, 9).setValues([[
      "'" + code, name, market, stype, ch, cathay, yuanta, fubon, kgi
    ]]);
    dstMain.getRange(targetRow, 17).setValue(Math.round(totCost));
    if (!priceMap[code] || !priceMap[code].price) {
      dstMain.getRange(targetRow, 12).setValue(inferredPrice);
    }
  });

  const srcDb = srcSs.getSheetByName('資料庫');
  const dbValues = srcDb.getRange(3, 2, Math.max(1, srcDb.getLastRow() - 2), 14).getValues();
  const activeDb = dbValues.filter(r => String(r[13]).trim() === '有效' && Number(r[8]) > 0);
  const dstBroker = dstSs.getSheetByName(CONFIG.BROKER_SHEET_NAME);

  activeDb.forEach((r, idx) => {
    const rowNum = 21 + idx;
    if (rowNum <= 93) {
      let code = String(r[1]).trim().replace(/\.0$/, '');
      if (['50', '56', '646', '662', '713', '878', '919'].indexOf(code) !== -1) {
        code = ('0000' + code).slice(-4);
      } else if (code === '9805') {
        code = '009805';
      }
      const broker = String(r[0]).trim();
      const name = String(r[2]).trim();
      const isOtc = CONFIG.OTC_CODES.indexOf(code) !== -1;
      const market = isOtc ? '上櫃 TPEx' : '上市 TWSE';
      const shares = Number(r[8]) || 0;
      const cost = Math.round(Number(r[9]) || 0);
      dstBroker.getRange(rowNum, 1, 1, 5).setValues([[broker, "'" + code, name, market, shares]]);
      dstBroker.getRange(rowNum, 7).setValue(cost);
    }
  });
}

/**
 * 三層免費即時報價引擎 (TWSE/TPEx MIS 批次 API + Yahoo Finance v8 API)
 */
function updateIntradayQuotes() {
  const dstSs = SpreadsheetApp.openById(CONFIG.STOCK_SPREADSHEET_ID);
  const dstMain = dstSs.getSheetByName(CONFIG.MAIN_SHEET_NAME);
  const dstBroker = dstSs.getSheetByName(CONFIG.BROKER_SHEET_NAME);

  const mainData = dstMain.getRange(8, 1, 43, 12).getValues();
  const codes = mainData.map(r => String(r[0]).trim().replace(/^'/, '')).filter(Boolean);
  const quotes = fetchAllQuotesMultiLayer(codes);

  const priceUpdates = mainData.map(r => {
    const code = String(r[0]).trim().replace(/^'/, '');
    const oldPrev = Number(r[10]) || 0;
    const oldPrice = Number(r[11]) || 0;
    const q = quotes[code];
    return [
      (q && q.prevClose > 0) ? q.prevClose : oldPrev,
      (q && q.price > 0) ? q.price : oldPrice
    ];
  });
  dstMain.getRange(8, 11, priceUpdates.length, 2).setValues(priceUpdates);

  const brokerData = dstBroker.getRange(21, 2, 73, 8).getValues();
  const brokerUpdates = brokerData.map(r => {
    const code = String(r[0]).trim().replace(/^'/, '');
    const oldPrev = Number(r[6]) || 0;
    const oldPrice = Number(r[7]) || 0;
    const q = quotes[code];
    return [
      (q && q.prevClose > 0) ? q.prevClose : oldPrev,
      (q && q.price > 0) ? q.price : oldPrice
    ];
  });
  dstBroker.getRange(21, 8, brokerUpdates.length, 2).setValues(brokerUpdates);
  SpreadsheetApp.flush();
}

function fetchAllQuotesMultiLayer(codes) {
  const result = {};
  const chunkSize = 25;

  for (let i = 0; i < codes.length; i += chunkSize) {
    const chunk = codes.slice(i, i + chunkSize);
    const exCh = chunk.map(c => {
      const prefix = CONFIG.OTC_CODES.indexOf(c) !== -1 ? 'otc_' : 'tse_';
      return prefix + c + '.tw';
    }).join('|');

    try {
      const url = 'https://mis.twse.com.tw/stock/api/getStockInfo.jsp?ex_ch=' + encodeURIComponent(exCh) + '&json=1&delay=0&_=' + Date.now();
      const resp = UrlFetchApp.fetch(url, {
        muteHttpExceptions: true,
        headers: { 'Accept-Language': 'zh-TW,zh;q=0.9', 'User-Agent': 'Mozilla/5.0' }
      });
      if (resp.getResponseCode() === 200) {
        const json = JSON.parse(resp.getContentText());
        if (json && Array.isArray(json.msgArray)) {
          json.msgArray.forEach(item => {
            const c = item.c;
            const prevClose = parseFloat(item.y) || 0;
            let price = parseFloat(item.z) || 0;
            if (!price && item.b && item.b !== '-') {
              price = parseFloat(String(item.b).split('_')[0]) || 0;
            }
            if (!price && item.a && item.a !== '-') {
              price = parseFloat(String(item.a).split('_')[0]) || 0;
            }
            if (!price && prevClose > 0) price = prevClose;
            if (c && (price > 0 || prevClose > 0)) {
              result[c] = { price: price, prevClose: prevClose };
            }
          });
        }
      }
    } catch (e) {}
  }

  codes.forEach(c => {
    if (!result[c] || !result[c].price || !result[c].prevClose) {
      const isOtc = CONFIG.OTC_CODES.indexOf(c) !== -1;
      const suffixes = isOtc ? ['.TWO', '.TW'] : ['.TW', '.TWO'];
      for (let s = 0; s < suffixes.length; s++) {
        try {
          const yUrl = 'https://query1.finance.yahoo.com/v8/finance/chart/' + c + suffixes[s] + '?interval=1m&range=1d';
          const r = UrlFetchApp.fetch(yUrl, { muteHttpExceptions: true });
          if (r.getResponseCode() === 200) {
            const data = JSON.parse(r.getContentText());
            const meta = data.chart && data.chart.result && data.chart.result[0] && data.chart.result[0].meta;
            if (meta && meta.regularMarketPrice) {
              result[c] = {
                price: Number(meta.regularMarketPrice),
                prevClose: Number(meta.chartPreviousClose || meta.previousClose || meta.regularMarketPrice)
              };
              break;
            }
          }
        } catch (err) {}
      }
    }
  });

  return result;
}

function scheduledMasterRunner() {
  const now = new Date();
  const day = now.getDay();
  const hour = now.getHours();
  const min = now.getMinutes();
  const hhmm = hour * 100 + min;

  if (day === 0 || day === 6) {
    if (hour === 20 && min < 15) {
      syncHoldingsFromMaster();
      updateIntradayQuotes();
    }
    return;
  }

  if (hhmm >= 825 && hhmm <= 835) {
    syncHoldingsFromMaster();
    updateIntradayQuotes();
    return;
  }

  if (hhmm >= 900 && hhmm <= 1335) {
    if (min < 5) {
      syncHoldingsFromMaster();
    }
    updateIntradayQuotes();
    return;
  }

  const postMarketHours = [15, 16, 17, 18, 20, 21, 22, 23];
  if (postMarketHours.indexOf(hour) !== -1 && min < 10) {
    syncHoldingsFromMaster();
    updateIntradayQuotes();
  }
}

function setupAllTriggers() {
  const triggers = ScriptApp.getProjectTriggers();
  triggers.forEach(t => {
    if (t.getHandlerFunction() === 'scheduledMasterRunner') {
      ScriptApp.deleteTrigger(t);
    }
  });

  ScriptApp.newTrigger('scheduledMasterRunner')
    .timeBased()
    .everyMinutes(5)
    .create();

  syncHoldingsFromMaster();
  updateIntradayQuotes();
}
